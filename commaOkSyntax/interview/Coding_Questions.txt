4. Count occurrences using a map, Using the comma-ok while updating/checking the map.

package main

import "fmt"

func main() {
	words := []string{
		"go", "java", "go", "python",
		"go", "java",
	}

	frequency := make(map[string]int)

	for _, word := range words {
		count, ok := frequency[word]

		if ok {
			frequency[word] = count + 1
		} else {
			frequency[word] = 1
		}
	}

	fmt.Println(frequency)
}

7. Handle multiple types
    values := []interface{}{
        "Golang",
        100,
        true,
        25.5,
    }

package main

import "fmt"

func main() {
	values := []interface{}{
		"Golang",
		100,
		true,
		25.5,
	}

	for _, value := range values {

		if v, ok := value.(string); ok {
			fmt.Println("String:", v)
		}

		if v, ok := value.(int); ok {
			fmt.Println("Integer:", v)
		}

		if v, ok := value.(bool); ok {
			fmt.Println("Boolean:", v)
		}

		if v, ok := value.(float64); ok {
			fmt.Println("Float:", v)
		}
	}
}
